\section{First Principles}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{What is a Phase Interpolator?}
  \begin{columns}[T]
    \begin{column}{0.46\linewidth}
      A circuit that produces a clock at \alert{any phase between two
      reference phases} by taking a weighted sum.

      \medskip
      \begin{itemize}
        \item \textbf{Inputs:} two reference clocks, same frequency,
              known phase offset (typically $90^\circ$ apart — I and Q)
        \item \textbf{Control:} digital code $\to$ analog weights
              $(\alpha, \beta)$
        \item \textbf{Output:} one clock with phase
              $\phi \in [\phi_I, \phi_Q]$, same frequency
      \end{itemize}

      \medskip
      No frequency synthesis, no PLL — \emph{just controlled mixing of
      two phases}.
    \end{column}
    \begin{column}{0.52\linewidth}
      \centering
      \fitwidth[\linewidth]{%
      \begin{tikzpicture}[scale=0.9, transform shape, thick, font=\small,
          port/.style={inner sep=1pt}]
        \draw[rounded corners, fill=uciSand, draw=uciNavy, thick]
            (0,0) rectangle (4.0,3.0);
        \node[uciNavy] at (2,2.6) {\textbf{Phase Interpolator}};
        \node[uciNavy, font=\scriptsize] at (2,2.0)
            {$v_{out}=\alpha\,v_I + \beta\,v_Q$};

        \draw[->, uciNavy] (-1.6,2.2) node[port,left]

            {$v_Q=\sin\omega t$} -- (0,0.8);

    \end{column}

\end{frame}

\begin{frame}{First Principles — I/Q Weighted Sum}
  \textbf{Given two orthogonal clocks at $\omega$:}
  \[
    v_I(t) = A\cos\omega t,\qquad
    v_Q(t) = A\sin\omega t.
  \]

  \textbf{Take a weighted sum} with real weights $\alpha, \beta$:
  \[
    v_{out}(t) = \alpha\,v_I(t) + \beta\,v_Q(t)
               = A\bigl[\alpha\cos\omega t + \beta\sin\omega t\bigr].
  \]

  \textbf{Apply the harmonic-addition identity}
  $a\cos\theta + b\sin\theta = R\cos(\theta-\phi)$ with
  $R=\sqrt{a^2+b^2}$, $\tan\phi = b/a$:
  \[
    \boxed{\;
      v_{out}(t)
        = A\sqrt{\alpha^2+\beta^2}\;
          \cos\!\left(\omega t - \phi\right),
      \quad
      \phi = \arctan\!\frac{\beta}{\alpha}.
    \;}
  \]

  \begin{block}{Takeaway}
    The output \emph{frequency} is unchanged. The output \emph{phase}
    is set entirely by the weight ratio $\beta/\alpha$ — sweep the
    weights, sweep the phase from $0^\circ$ (pure I) to $90^\circ$
    (pure Q).
  \end{block}
\end{frame}

    \begin{column}{0.43\linewidth}
      \textbf{Geometry = algebra:}
      \begin{itemize}
        \item $\alpha$ slides the tip along $I$
        \item $\beta$ slides the tip along $Q$
        \item Tip's polar angle \emph{is} $\phi$
      \end{itemize}

      \medskip
      \textbf{Two practical knobs:}
      \begin{description}\setlength\itemsep{2pt}
        \item[\alert{Quadrant select}] choose which two of
              $\{I, Q, \overline{I}, \overline{Q}\}$ to sum
              $\to$ $360^\circ$ coverage (coarse)
        \item[\alert{Weight ratio}] $N$-bit DAC sets $\beta/\alpha$
              $\to$ $2^N$ phase steps inside the quadrant (fine)
      \end{description}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Application 1 — Wireline / SerDes CDR}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \textbf{Problem.} Receiver gets serial data with no clock. Local
      reference clock is at the right \emph{frequency} but unknown
      \emph{phase}.

      \medskip
      \textbf{PI's job.} Slide the sampling clock until it lands in the
      \emph{centre of the data eye} — and keep it there as the link
      drifts.
      \[
        \phi_{samp}
          = \tfrac{1}{2}\bigl(\phi_{rise}+\phi_{fall}\bigr)
      \]

      \begin{block}{Why a PI?}
        Resolution of $360^\circ/2^N$ at multi-GHz rates with no PLL in
        the loop. A bang-bang phase detector dithers the PI code
        $\pm 1$ LSB at lock — the entire CDR loop is digital after the
        sampler.
      \end{block}
    \end{column}

\begin{frame}{Application 2 — Optical Coherent Receivers}
  \begin{columns}[T]
    \begin{column}{0.50\linewidth}
      \textbf{Setting.} 100--800 Gb/s coherent links: optical front end
      delivers I and Q baseband (after mixing with a local laser) into
      ADCs running at \SI{50}{\giga S/s}+.

      \medskip
      \textbf{Where the PI lives.}
      \begin{itemize}
        \item Generates the \emph{ADC sampling clock phase} per lane
        \item Tracks slow phase drift between TX and RX symbol clocks
              with no analog VCO retuning
        \item Per-lane PI codes equalise inter-lane skew on parallel
              DSP fabric
      \end{itemize}

        Clock jitter degrades SNR for 16/64-QAM. A $7$-bit PI gives
        $\Delta\phi \approx 2.8^\circ \approx \SI{80}{\femto\second}$

        }

        \draw[<->, red, thick] (0.25,2.6) -- ++({90-\theta}:0.8);
        \node[red, right=2pt] at (0.55,2.9) {$d\sin\theta$};

        \node[uciNavy, font=\scriptsize] at (3.0,-0.9)
          {progressive delay $\Delta t = (d\sin\theta)/c$};
        \node[uciNavy, font=\scriptsize] at (3.0,-1.3)
          {$\Rightarrow$ progressive phase
           $\Delta\phi = \tfrac{2\pi d}{\lambda}\sin\theta$};

        \node[draw=uciNavy, fill=uciGoldPale, rounded corners,
              font=\tiny, align=center] at (5.5,1.3)
          {one PI per element\\ provides $\Delta\phi$};
      \end{tikzpicture}}
    \end{column}
    \begin{column}{0.43\linewidth}
      \textbf{Idea.} Feed each antenna the \emph{same signal} but with
      a progressive phase shift $\Delta\phi$ between neighbours. The
      array radiates a beam at angle
      \[
        \boxed{\;\sin\theta
          = \frac{\lambda}{2\pi d}\,\Delta\phi.\;}
      \]
      Change $\Delta\phi$ electronically $\to$ steer $\theta$
      with no moving parts.

      \medskip
      \textbf{Why a PI per element?}
      \begin{itemize}
        \item Continuous, monotonic phase
        \item Per-element skew calibration
        \item Same hardware across the band — only $\Delta\phi$ changes
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phase Resolution $\to$ Beam-Steering Resolution}
  \textbf{Array equation} ($d$ spacing, $\lambda$ wavelength).
  Differentiate to map small $\Delta\phi$ to small $\Delta\theta$:
  \[
    \Delta\phi = \tfrac{2\pi d}{\lambda}\sin\theta
    \;\Longrightarrow\;
    \boxed{\;\Delta\theta
      = \frac{\lambda}{2\pi d\cos\theta}\,\Delta\phi_{\min}.\;}
  \]

  For an $N$-bit PI ($\Delta\phi_{\min} = 2\pi/2^N$) at the typical
  half-wavelength spacing $d=\lambda/2$:
  $\Delta\theta = (1/\cos\theta)\cdot 2^{-(N-1)}\,$rad.

  \begin{columns}[T]
    \begin{column}{0.55\linewidth}
      \centering\footnotesize
      \begin{tabular}{@{}cccc@{}}
        \toprule
        $N$ & $\Delta\phi_{\min}$ & $\Delta\theta_{\,@\,0^\circ}$
                                  & $\Delta\theta_{\,@\,60^\circ}$ \\
        \midrule
        4 & $22.5^\circ$ & $7.2^\circ$  & $14.3^\circ$ \\
        6 & $5.6^\circ$  & $1.8^\circ$  & $3.6^\circ$  \\
        8 & $1.4^\circ$  & $0.45^\circ$ & $0.90^\circ$ \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.43\linewidth}
      \begin{alertblock}{Two asymmetries}
        $1/\cos\theta$ blows up near endfire — same LSB buys coarser
        steps off-broadside. And PI \emph{nonlinearity} (DNL/INL) sets
        the worst-case step, not the average.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}\setlength\itemsep{4pt}
    \item A PI is a \alert{weighted sum of two reference phases}; the
          weight ratio sets $\phi = \arctan(\beta/\alpha)$, the
          frequency is unchanged.
    \item Geometry is just the phasor tip sliding inside an I/Q
          quadrant; coarse = quadrant select, fine = $N$-bit weight
          DAC.
    \item Realised as two CML pairs sharing a load — current-domain
          summation, no analog multiplier.
    \item \textbf{SerDes / optical:} sets the data sampling clock
          phase in fully digital CDR loops at \si{\giga Hz}+ rates.
    \item \textbf{Phased arrays:} sets per-element progressive phase;
          $\Delta\theta = \frac{\lambda}{2\pi d\cos\theta}\Delta\phi$
          ties phase resolution to angular resolution.
    \item Bit-count of the PI is the lever shared by every
          application — more bits buy finer eye-centering, lower DSP
          jitter, finer beam steps.
  \end{itemize}
\end{frame}
