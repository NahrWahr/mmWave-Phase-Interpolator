\begin{frame}{Architecture Context}
\begin{columns}[T]
  \begin{column}{0.52\linewidth}
    \textbf{This work:} \SI{35}{\giga\hertz} half-rate PI\,$\rightarrow$\,$\times2$
    doubler\,$\rightarrow$\,\SI{70}{\giga\hertz}\\[1ex]
    Phase-error doubling rule:
    \[
      \epsilon_{\Phi,70} = 2\,\epsilon_{\phi,35}
    \]
    {\small Phase noise $+6\,\mathrm{dB}$ ($20\log_{10}2$); absolute jitter preserved.}
  \end{column}
  \begin{column}{0.44\linewidth}
    \begin{block}{Baseline families — 8 works}
    \begin{itemize}\small
      \item Active vector-modulator PS (15--90~GHz)
      \item Passive RTPS (28~GHz)
      \item Passive switched APN (14--50~GHz)
      \item Passive PI in T/R channels (W-band)
    \end{itemize}
    \end{block}
  \end{column}
\end{columns}
\end{frame}

% ── Win Conditions ─────────────────────────────────────────────────────────
\begin{frame}{Win Conditions}
\begin{alertblock}{vs.\ active vector-modulator baselines}
\begin{itemize}
  \item \textbf{W-band VM (Pepe):}
    $\epsilon_{\phi,35} < 4.5^\circ$ $+$ low doubler AM-PM
    $\Rightarrow$ $\epsilon_{\Phi,70} < 9^\circ$ (beats Pepe PS1/PS2)
